Collections (Map, Set, WeakMap, WeakSet)

Import:

import { Map, Set } from "@tsonic/js/index.js";

Example:

import { console, Map, Set } from "@tsonic/js/index.js";

export function main(): void {
  const m = new Map<string, number>();
  m.set("a", 1);
  console.log(m.get("a"));

  const s = new Set<number>();
  s.add(1);
  console.log(s.has(1));
}
