RegExp

Import:

import { RegExp } from "@tsonic/js/index.js";

Example:

import { console, RegExp } from "@tsonic/js/index.js";

export function main(): void {
  const re = new RegExp("a+", "g");
  console.log(re.test("caaab"));
}
